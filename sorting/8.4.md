# 8.4: Bucket Sort
## 8.4-1: Bucket Sort Operation
<pre>
Bucket Sort Procedure
BUCKET-SORT(A, n)
    let B[0:n-1] be a new array
    for i: 0 to n-1
        make B[i] an empty list
    for i: 1 to n
        insert A[i] into B[floor(n * A[i])]
    for i: 0 to n-1
        sort list B[i] with insertion sort
    concatenate the lists B[0], B[1], ... ,B[n-1] in order
    return concatenated list

Given Array
    A = [.79, .13, .16, .64, .39, .20, .89, .53, .71, .42]
    , n = 10
Let B = [[], [], [], [], [], [], [], [], [], []]    (: list with n empty lists/arrays)
Inserting Elements
    A[1] = .79 => B[floor(10 * .79)] = B[7]
        => B =[[], [], [], [], [], [], [.79], [], [], []]
    A[2] = .13 => B[floor(10 * .13)] = B[1]
        => B =[[.13], [], [], [], [], [], [.79], [], [], []]
    A[3] = .16 => B[floor(10 * .16)] = B[1]
        => B =[[.13, .16], [], [], [], [], [], [.79], [], [], []]
    A[4] = .64 => B[floor(10 * .64)] = B[6]
        => B =[[.13, .16], [], [], [], [], [.64], [.79], [], [], []]
    A[5] = .39 => B[floor(10 * .39)] = B[3]
        => B =[[.13, .16], [], [.39], [], [], [.64], [.79], [], [], []]
    A[6] = .20 => B[floor(10 * .20)] = B[2]
        => B =[[.13, .16], [.20], [.39], [], [], [.64], [.79], [], [], []]
    A[7] = .89 => B[floor(10 * .89)] = B[8]
        => B =[[.13, .16], [.20], [.39], [], [], [.64], [.79], [.89], [], []]
    A[8] = .53 => B[floor(10 * .53)] = B[5]
        => B =[[.13, .16], [.20], [.39], [], [.53], [.64], [.79], [.89], [], []]
    A[9] = .71 => B[floor(10 * .71)] = B[7]
        => B =[[.13, .16], [.20], [.39], [], [.53], [.64], [.79, .71], [.89], [], []]
    A[910] = .42 => B[floor(10 * .42)] = B[4]
        => B =[[.13, .16], [.20], [.39], [.42], [.53], [.64], [.79, .71], [.89], [], []]

Hence we get
    B = [[.13, .16], [.20], [.39], [.42], [.53], [.64], [.79, .71], [.89], [], []]
Sorting each individual list
    B = [[.13, .16], [.20], [.39], [.42], [.53], [.64], [.71, .79], [.89], [], []]
Concatenating
    Sorted Array = [.13, .16, .20, .39, .42, .53, .64, .71, .79, .89]
